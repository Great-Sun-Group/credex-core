Recurring Module

Key features:

  • Owner-only modification rights
  • Schedule validation checks
  • Active status validations
  • Auto-acceptance for special templates
  • Tier-based subscription management

Template Types:

  • REGULAR: Standard recurring payments with flexible frequency
  • DCO_GIVE: Daily credex offerings to foundation accounts
  • MEMBERTIER_SUBSCRIPTION: Member tier subscription payments
    • Fixed 28-day payment cycle
    • Secured USD credex only
    • Auto-acceptance flow
    • Tier-specific amount requirements
// Example from subscription validation
function validateSubscriptionTemplate(data) {
  if (data.memberTier !== 3) {
    return { isValid: false, message: 'Currently only tier 3 is supported' };
  }
  if (data.payFrequency !== 28) {
    return { isValid: false, message: 'Subscription payment frequency must be 28 days' };
  }
  if (!data.securedCredex) {
    return { isValid: false, message: 'Subscription payments must use secured credex' };
  }
  return { isValid: true, message: 'Valid subscription template' };
}

Recurring Module

// Core relationships - Template Creation
MATCH (sourceAccount:Account)-[:SCHEDULES]->(recurring:Recurring)
  -[:TARGETS]->(targetAccount:Account)

// State Management
MATCH (recurring)-[:REQUESTS]->(targetAccount:Account)  // Pending state
MATCH (recurring)-[:ACTIVE]->(targetAccount:Account)    // Accepted state

// Status Tracking
MATCH (recurring {status: "ACTIVE"})  // Active subscriptions
MATCH (recurring {status: "PENDING"}) // Pending subscriptions
MATCH (recurring {status: "CANCELLED"}) // Cancelled subscriptions

// Subscription-specific
MATCH (member:Member)-[OWNS]->(account:Account)
  -[:ACTIVE]->(subscription:Recurring { 
    templateType: "MEMBERTIER_SUBSCRIPTION" 
  })-[:SIGNED]->(memberTierPayment:Credex)
  -[:CREATED_ON]->(daynode:Daynode)
WHERE daynode.Date >= date() - duration('P28D')

// Auto-acceptance Flow
MATCH (recurring:Recurring {
  templateType: "MEMBERTIER_SUBSCRIPTION",
  status: "PENDING"
})
SET recurring.status = "ACTIVE",
    recurring.acceptedAt = datetime()

// Payment Processing
MATCH (subscription:Recurring {
  templateType: "MEMBERTIER_SUBSCRIPTION",
  status: "ACTIVE"
})
WHERE date(subscription.nextPayDate) <= date()
SET subscription.nextPayDate = date() + duration('P28D')

// Failed Payment Handling
MATCH (subscription:Recurring {
  templateType: "MEMBERTIER_SUBSCRIPTION",
  status: "ACTIVE"
})
WHERE subscription.lastPaymentFailed = true
SET subscription.nextPayDate = date() + duration('P1D')
